(%- for p in persons %)
(%-   if p.team|length > 0 %)
(%-     do p.update({"sort_team":p.team[0][0]}) %)
(%-   else %)
(%-     do p.update({"sort_team":""}) %)
(%-   endif %)
(%- endfor %)

(%- set teams = [] %)
(%- for p in persons %)
(%-   if not p.sort_team in teams %)
(%-     set name = p.sort_team %)
(%-     do teams.append(name)%)
(%-   endif %)
(%- endfor %)

(%- for team in teams|sort %)
(%-   set size = 100 %)
(%-   if team|length > 10 %)
(%-   set size = 80 %)
(%-   endif %)
(( namesign(team, size) ))
(% endfor %)
